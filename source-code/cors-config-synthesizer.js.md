/\*\*  
 \* CORSHeaderForge Pro \- Multi-Server Configuration Synthesizer  
 \* Generates production-ready configurations for Nginx, Apache, Cloudflare Workers, Caddy, CloudFront Functions, and Next.js.  
 \*/

class CORSConfigSynthesizer {  
  constructor() {  
    this.engine \= typeof CORSHeaderEngine \!== 'undefined' ? new CORSHeaderEngine() : null;  
  }

  synthesize(profile, targetServer \= 'nginx') {  
    if (\!this.engine && typeof require \!== 'undefined') {  
      const EngineClass \= require('./cors-header-engine.js');  
      this.engine \= new EngineClass();  
    }  
    const headersMap \= this.engine.buildHeadersMap(profile);  
    const cors \= profile.cors || {};  
    const origins \= cors.allowedOrigins || \[\];  
    const allowCreds \= \!\!cors.allowCredentials;

    switch (targetServer.toLowerCase()) {  
      case 'nginx':  
        return this.generateNginx(profile, headersMap);  
      case 'apache':  
        return this.generateApache(profile, headersMap);  
      case 'cloudflare':  
        return this.generateCloudflareWorker(profile, headersMap);  
      case 'caddy':  
        return this.generateCaddy(profile, headersMap);  
      case 'cloudfront':  
        return this.generateCloudFrontFunction(profile, headersMap);  
      case 'nextjs':  
        return this.generateNextJs(profile, headersMap);  
      default:  
        return JSON.stringify(headersMap, null, 2);  
    }  
  }

  generateNginx(profile, headersMap) {  
    const cors \= profile.cors || {};  
    const origins \= cors.allowedOrigins || \[\];  
    const allowCreds \= \!\!cors.allowCredentials;

    const lines \= \[\];  
    lines.push(\`\# Nginx Production Security Headers & CORS Configuration\`);  
    lines.push(\`\# Generated by CORSHeaderForge Pro for: \${profile.domain || 'api.example.com'}\`);  
    lines.push(\`\`);

    if (origins.length \> 1 && \!origins.includes('\*')) {  
      lines.push(\`\# Validate Origin dynamically against trusted allowlist\`);  
      lines.push(\`map \$http\_origin \$cors\_allowed\_origin {\`);  
      lines.push(\`    default "";\`);  
      origins.forEach(orig \=\> {  
        lines.push(\`    "\~^\${orig.replace(/\\./g, '\\\\.')}\$" "\$http\_origin";\`);  
      });  
      lines.push(\`}\`);  
      lines.push(\`\`);  
    }

    lines.push(\`server {\`);  
    lines.push(\`    listen 443 ssl http2;\`);  
    lines.push(\`    server\_name \${profile.domain || 'api.example.com'};\`);  
    lines.push(\`\`);  
    lines.push(\`    \# CORS Preflight Handler\`);  
    lines.push(\`    if (\$request\_method \= 'OPTIONS') {\`);  
    if (origins.length \=== 1 && origins\[0\] \!== '\*') {  
      lines.push(\`        add\_header 'Access-Control-Allow-Origin' '\${origins\[0\]}' always;\`);  
    } else if (origins.includes('\*')) {  
      lines.push(\`        add\_header 'Access-Control-Allow-Origin' '\*' always;\`);  
    } else {  
      lines.push(\`        add\_header 'Access-Control-Allow-Origin' \$cors\_allowed\_origin always;\`);  
    }  
    if (cors.allowedMethods && cors.allowedMethods.length \> 0\) {  
      lines.push(\`        add\_header 'Access-Control-Allow-Methods' '\${cors.allowedMethods.join(', ')}' always;\`);  
    }  
    if (cors.allowedHeaders && cors.allowedHeaders.length \> 0\) {  
      lines.push(\`        add\_header 'Access-Control-Allow-Headers' '\${cors.allowedHeaders.join(', ')}' always;\`);  
    }  
    if (cors.exposedHeaders && cors.exposedHeaders.length \> 0\) {  
      lines.push(\`        add\_header 'Access-Control-Expose-Headers' '\${cors.exposedHeaders.join(', ')}' always;\`);  
    }  
    if (typeof cors.maxAge \=== 'number') {  
      lines.push(\`        add\_header 'Access-Control-Max-Age' \${cors.maxAge} always;\`);  
    }  
    if (allowCreds) {  
      lines.push(\`        add\_header 'Access-Control-Allow-Credentials' 'true' always;\`);  
    }  
    lines.push(\`        add\_header 'Content-Type' 'text/plain; charset=utf-8';\`);  
    lines.push(\`        add\_header 'Content-Length' 0;\`);  
    lines.push(\`        return 204;\`);  
    lines.push(\`    }\`);  
    lines.push(\`\`);

    lines.push(\`    \# Production HTTP Security & CORS Response Headers\`);  
    for (const \[key, val\] of Object.entries(headersMap)) {  
      if (key.startsWith('Access-Control-Allow-Origin')) {  
        if (origins.length \> 1 && \!origins.includes('\*')) {  
          lines.push(\`    add\_header \${key} \$cors\_allowed\_origin always;\`);  
        } else {  
          lines.push(\`    add\_header \${key} "\${val}" always;\`);  
        }  
      } else {  
        lines.push(\`    add\_header \${key} "\${val.replace(/"/g, '\\\\"')}" always;\`);  
      }  
    }

    lines.push(\`\`);  
    lines.push(\`    location / {\`);  
    lines.push(\`        try\_files \$uri \$uri/ /index.html;\`);  
    lines.push(\`    }\`);  
    lines.push(\`}\`);

    return lines.join('\\n');  
  }

  generateApache(profile, headersMap) {  
    const cors \= profile.cors || {};  
    const origins \= cors.allowedOrigins || \[\];  
    const lines \= \[\];

    lines.push(\`\# Apache HTTP Server / .htaccess Configuration\`);  
    lines.push(\`\# Generated by CORSHeaderForge Pro for: \${profile.domain || 'api.example.com'}\`);  
    lines.push(\`\`);  
    lines.push(\`\<IfModule mod\_headers.c\>\`);

    for (const \[key, val\] of Object.entries(headersMap)) {  
      lines.push(\`    Header always set \${key} "\${val.replace(/"/g, '\\\\"')}"\`);  
    }

    lines.push(\`\</IfModule\>\`);  
    lines.push(\`\`);  
    lines.push(\`\<IfModule mod\_rewrite.c\>\`);  
    lines.push(\`    RewriteEngine On\`);  
    lines.push(\`    RewriteCond %{REQUEST\_METHOD} OPTIONS\`);  
    lines.push(\`    RewriteRule ^(.\*)\$ \$1 \[R=204,L\]\`);  
    lines.push(\`\</IfModule\>\`);

    return lines.join('\\n');  
  }

  generateCloudflareWorker(profile, headersMap) {  
    const cors \= profile.cors || {};  
    const origins \= cors.allowedOrigins || \[\];  
    const allowCreds \= \!\!cors.allowCredentials;

    return \`/\*\*  
 \* Cloudflare Worker \- Zero-Trust CORS & Security Headers Middleware  
 \* Generated by CORSHeaderForge Pro for: \${profile.domain || 'api.example.com'}  
 \*/

const ALLOWED\_ORIGINS \= \${JSON.stringify(origins, null, 2)};

export default {  
  async fetch(request, env, ctx) {  
    const origin \= request.headers.get("Origin");  
    const isAllowedOrigin \= ALLOWED\_ORIGINS.includes("\*") || (origin && ALLOWED\_ORIGINS.includes(origin));

    // Handle CORS Preflight OPTIONS  
    if (request.method \=== "OPTIONS") {  
      const preflightHeaders \= new Headers();  
      if (isAllowedOrigin) {  
        preflightHeaders.set("Access-Control-Allow-Origin", origin || "\*");  
      }  
      preflightHeaders.set("Access-Control-Allow-Methods", "\${(cors.allowedMethods || \['GET', 'POST', 'OPTIONS'\]).join(', ')}");  
      preflightHeaders.set("Access-Control-Allow-Headers", "\${(cors.allowedHeaders || \['Content-Type', 'Authorization'\]).join(', ')}");  
      preflightHeaders.set("Access-Control-Max-Age", "\${cors.maxAge || 86400}");  
      \${allowCreds ? 'preflightHeaders.set("Access-Control-Allow-Credentials", "true");' : ''}  
      return new Response(null, { status: 204, headers: preflightHeaders });  
    }

    // Forward actual request to upstream origin  
    const response \= await fetch(request);  
    const newHeaders \= new Headers(response.headers);

    // Inject CORS Headers  
    if (isAllowedOrigin) {  
      newHeaders.set("Access-Control-Allow-Origin", origin || "\*");  
      \${allowCreds ? 'newHeaders.set("Access-Control-Allow-Credentials", "true");' : ''}  
    }

    // Inject OWASP Web Security Headers  
\${Object.entries(headersMap)  
  .filter((\[k\]) \=\> \!k.startsWith('Access-Control-'))  
  .map((\[k, v\]) \=\> \`    newHeaders.set(\${JSON.stringify(k)}, \${JSON.stringify(v)});\`)  
  .join('\\n')}

    return new Response(response.body, {  
      status: response.status,  
      statusText: response.statusText,  
      headers: newHeaders  
    });  
  }  
};\`;  
  }

  generateCaddy(profile, headersMap) {  
    const lines \= \[\];  
    lines.push(\`\# Caddyfile Configuration\`);  
    lines.push(\`\# Generated by CORSHeaderForge Pro for: \${profile.domain || 'api.example.com'}\`);  
    lines.push(\`\`);  
    lines.push(\`\${profile.domain || 'api.example.com'} {\`);  
    lines.push(\`    \# Security Headers\`);  
    lines.push(\`    header {\`);  
    for (const \[key, val\] of Object.entries(headersMap)) {  
      lines.push(\`        \${key} "\${val.replace(/"/g, '\\\\"')}"\`);  
    }  
    lines.push(\`    }\`);  
    lines.push(\`\`);  
    lines.push(\`    \# CORS Preflight Fast-Path\`);  
    lines.push(\`    @options {\`);  
    lines.push(\`        method OPTIONS\`);  
    lines.push(\`    }\`);  
    lines.push(\`    respond @options 204\`);  
    lines.push(\`\`);  
    lines.push(\`    reverse\_proxy localhost:8080\`);  
    lines.push(\`}\`);

    return lines.join('\\n');  
  }

  generateCloudFrontFunction(profile, headersMap) {  
    return \`/\*\*  
 \* AWS CloudFront Function \- Viewer Response Event  
 \* Sub-millisecond Security Header & CORS Injector  
 \* Generated by CORSHeaderForge Pro  
 \*/

function handler(event) {  
    var response \= event.response;  
    var headers \= response.headers;

\${Object.entries(headersMap).map((\[k, v\]) \=\> {  
  const lk \= k.toLowerCase();  
  return \`    headers\['\${lk}'\] \= { value: '\${v.replace(/'/g, "\\\\'")}' };\`;  
}).join('\\n')}

    return response;  
}\`;  
  }

  generateNextJs(profile, headersMap) {  
    const headerObjs \= Object.entries(headersMap).map((\[key, value\]) \=\> ({ key, value }));

    return \`// next.config.js \- Security & CORS Headers  
// Generated by CORSHeaderForge Pro for: \${profile.domain || 'api.example.com'}

module.exports \= {  
  async headers() {  
    return \[  
      {  
        source: '/:path\*',  
        headers: \${JSON.stringify(headerObjs, null, 10)}  
      }  
    \];  
  }  
};\`;  
  }  
}

// Module export for Node.js & browser  
if (typeof module \!== 'undefined' && module.exports) {  
  module.exports \= CORSConfigSynthesizer;  
} else if (typeof window \!== 'undefined') {  
  window.CORSConfigSynthesizer \= CORSConfigSynthesizer;  
}
